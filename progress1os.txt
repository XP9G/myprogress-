<script>

/* =========================
   API
========================= */

const API = "https://progress-os-api.aoart20.workers.dev";

let DATA = {
    dailyTasks: [],
    weeklyTasks: [],
    goals: [],
    achievements: [],
    timetable: [],
    reviews: []
};

/* =========================
   API REQUEST
========================= */

async function api(path, options = {}) {

    const config = {
        credentials: "include",
        ...options,
        headers: {
            "Content-Type": "application/json",
            ...(options.headers || {})
        }
    };

    const response = await fetch(API + path, config);

    let data;

    try {
        data = await response.json();
    } catch {
        throw new Error("Invalid server response");
    }

    if (!response.ok) {
        throw new Error(data.error || "Request failed");
    }

    return data;
}

/* =========================
   LOGIN
========================= */

async function login() {

    const username =
        document.getElementById("username").value.trim();

    const password =
        document.getElementById("password").value;

    const error =
        document.getElementById("loginError");

    error.textContent = "";

    if (!username || !password) {
        error.textContent =
            "Enter username and password.";
        return;
    }

    try {

        await api("/api/login", {
            method: "POST",
            body: JSON.stringify({
                username,
                password
            })
        });

        document
            .getElementById("loginScreen")
            .classList.add("hidden");

        document
            .getElementById("app")
            .classList.remove("hidden");

        await loadData();

    } catch (err) {

        error.textContent =
            err.message || "Login failed.";

    }
}

/* =========================
   LOGOUT
========================= */

async function logout() {

    try {
        await api("/api/logout", {
            method: "POST"
        });
    } catch {}

    document
        .getElementById("app")
        .classList.add("hidden");

    document
        .getElementById("loginScreen")
        .classList.remove("hidden");

    document.getElementById("password").value = "";
}

/* =========================
   LOAD DATA
========================= */

async function loadData() {

    try {

        const result =
            await api("/api/data");

        DATA.dailyTasks =
            result.dailyTasks || [];

        DATA.weeklyTasks =
            result.weeklyTasks || [];

        DATA.goals =
            result.goals || [];

        DATA.achievements =
            result.achievements || [];

        DATA.timetable =
            result.timetable || [];

        DATA.reviews =
            result.reviews || [];

        renderAll();

    } catch (err) {

        console.error(err);

        if (err.message === "Unauthorized") {

            document
                .getElementById("app")
                .classList.add("hidden");

            document
                .getElementById("loginScreen")
                .classList.remove("hidden");
        }
    }
}

/* =========================
   RENDER ALL
========================= */

function renderAll() {

    renderDaily();
    renderWeekly();
    renderGoals();
    renderAchievements();
    renderTimetable();

    updateDashboard();
    updateProgress();
}

/* =========================
   NAVIGATION
========================= */

function showSection(id, button) {

    document
        .querySelectorAll(".section")
        .forEach(section => {
            section.classList.remove("active");
        });

    document
        .getElementById(id)
        .classList.add("active");

    document
        .querySelectorAll(".nav-btn")
        .forEach(btn => {
            btn.classList.remove("active");
        });

    button.classList.add("active");
}

/* =========================
   DAILY
========================= */

async function addDaily() {

    const input =
        document.getElementById("dailyInput");

    const title = input.value.trim();

    if (!title) return;

    try {

        await api("/api/daily", {
            method: "POST",
            body: JSON.stringify({ title })
        });

        input.value = "";

        await loadData();

    } catch (err) {

        alert(err.message);

    }
}

async function toggleDaily(id, completed) {

    try {

        await api("/api/daily", {
            method: "PUT",
            body: JSON.stringify({
                id,
                completed: !completed
            })
        });

        await loadData();

    } catch (err) {

        alert(err.message);

    }
}

async function deleteDaily(id) {

    if (!confirm("Delete this task?")) return;

    try {

        await api("/api/daily", {
            method: "DELETE",
            body: JSON.stringify({ id })
        });

        await loadData();

    } catch (err) {

        alert(err.message);

    }
}

function renderDaily() {

    const list =
        document.getElementById("dailyList");

    list.innerHTML = "";

    if (DATA.dailyTasks.length === 0) {

        list.innerHTML =
            "<p>No daily tasks yet.</p>";

        return;
    }

    DATA.dailyTasks.forEach(task => {

        const item =
            document.createElement("div");

        item.className = "item";

        item.innerHTML = `
            <input
                type="checkbox"
                ${task.completed ? "checked" : ""}
                onchange="toggleDaily(${task.id}, ${!!task.completed})"
            >

            <div class="item-text ${
                task.completed ? "completed" : ""
            }">
                ${escapeHTML(task.title)}
            </div>

            <button
                class="delete-btn"
                onclick="deleteDaily(${task.id})"
            >
                Delete
            </button>
        `;

        list.appendChild(item);

    });
}

/* =========================
   GOALS
========================= */

async function addGoal() {

    const title =
        document.getElementById("goalInput")
        .value.trim();

    const progress =
        Number(
            document.getElementById("goalProgress")
            .value
        );

    if (!title) return;

    try {

        await api("/api/goals", {
            method: "POST",
            body: JSON.stringify({
                title,
                progress: Math.max(
                    0,
                    Math.min(100, progress || 0)
                )
            })
        });

        document.getElementById("goalInput").value = "";
        document.getElementById("goalProgress").value = 0;

        await loadData();

    } catch (err) {

        alert(err.message);

    }
}

async function updateGoal(id, title, progress) {

    try {

        await api("/api/goals", {
            method: "PUT",
            body: JSON.stringify({
                id,
                title,
                progress
            })
        });

        await loadData();

    } catch (err) {

        alert(err.message);

    }
}

async function deleteGoal(id) {

    if (!confirm("Delete this goal?")) return;

    try {

        await api("/api/goals", {
            method: "DELETE",
            body: JSON.stringify({ id })
        });

        await loadData();

    } catch (err) {

        alert(err.message);

    }
}

function renderGoals() {

    const container =
        document.getElementById("goalsList");

    container.innerHTML = "";

    if (DATA.goals.length === 0) {

        container.innerHTML =
            "<p>No goals yet.</p>";

        return;
    }

    DATA.goals.forEach(goal => {

        const div =
            document.createElement("div");

        div.className = "goal";

        const progress =
            Math.max(
                0,
                Math.min(100, Number(goal.progress) || 0)
            );

        div.innerHTML = `

            <strong>
                ${escapeHTML(goal.title)}
            </strong>

            <div class="progress-bar">
                <div
                    class="progress-fill"
                    style="width:${progress}%"
                ></div>
            </div>

            <div style="display:flex;gap:8px;align-items:center;">

                <input
                    type="number"
                    min="0"
                    max="100"
                    value="${progress}"
                    id="goal-${goal.id}"
                    style="width:90px;padding:7px;"
                >

                <button
                    class="small-btn"
                    onclick="
                        updateGoal(
                            ${goal.id},
                            '${escapeJS(goal.title)}',
                            Number(document.getElementById('goal-${goal.id}').value)
                        )
                    "
                >
                    Update
                </button>

                <button
                    class="delete-btn"
                    onclick="deleteGoal(${goal.id})"
                >
                    Delete
                </button>

            </div>
        `;

        container.appendChild(div);

    });
}

async function saveTimetable() {

    const day =
        document.getElementById("timeDay").value;

    const time =
        document.getElementById("timeTime")
        .value.trim();

    const activity =
        document.getElementById("timeActivity")
        .value.trim();

    if (!time || !activity) return;

    try {

        await api("/api/timetable", {
            method: "POST",
            body: JSON.stringify({
                day,
                time,
                activity
            })
        });

        document.getElementById("timeTime").value = "";
        document.getElementById("timeActivity").value = "";

        await loadData();

    } catch (err) {

        alert(err.message);

    }
}

/* =========================
   PROGRESS
========================= */

function updateProgress() {

    const dailyCompleted =
        DATA.dailyTasks.filter(
            task => Number(task.completed) === 1
        ).length;

    const weeklyCompleted =
        DATA.weeklyTasks.filter(
            task => Number(task.completed) === 1
        ).length;

    let average = 0;

    if (DATA.goals.length > 0) {

        average =
            DATA.goals.reduce(
                (sum, goal) =>
                    sum + Number(goal.progress || 0),
                0
            ) / DATA.goals.length;

        average = Math.round(average);
    }

    document.getElementById("completedDaily")
        .textContent =
        dailyCompleted;

    document.getElementById("completedWeekly")
        .textContent =
        weeklyCompleted;

    document.getElementById("averageGoal")
        .textContent =
        average + "%";
}

/* =========================
   SECURITY / HTML ESCAPE
========================= */

function escapeHTML(value) {

    return String(value)
        .replace(/&/g, "&amp;")
        .replace(/</g, "&lt;")
        .replace(/>/g, "&gt;")
        .replace(/"/g, "&quot;")
        .replace(/'/g, "&#039;");
}

function escapeJS(value) {

    return String(value)
        .replace(/\\/g, "\\\\")
        .replace(/'/g, "\\'")
        .replace(/\n/g, "\\n")
        .replace(/\r/g, "\\r");
}
